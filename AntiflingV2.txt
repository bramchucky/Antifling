local SOUND_ON     = "rbxasset://sounds/electronicpingshort.wav"

local function playSound(soundId, pitch, volume)
    local s = Instance.new("Sound")
    s.SoundId = soundId
    s.Volume = volume or 0.5
    s.PlaybackSpeed = pitch or 1

    local ok = pcall(function()
        SoundService:PlayLocalSound(s)
    end)

    if not ok then
        pcall(function()
            s.Parent = SoundService
            s:Play()
        end)
    end

    Debris:AddItem(s, 3)
end

local function playToggleSound(state)
    if state then
        playSound(SOUND_ON, 1.15, 0.55)
    else
        playSound(SOUND_OFF, 0.9, 0.5)
    end
end

local function playMasterSound(state)
    if state then
        playSound(SOUND_ON, 1.35, 0.65)
        task.delay(0.08, function()
            playSound(SOUND_ON, 1.6, 0.45)
        end)
    else
        playSound(SOUND_OFF, 0.75, 0.55)
    end
end

local function playMinMaxSound(isOpening)
    playSound(SOUND_MINMAX, isOpening and 1.2 or 0.85, 0.45)
end

--// ================= HELPER KARAKTER (selalu valid) =================

local function getCharParts()
    local char = LocalPlayer.Character
    if not char or not char.Parent then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not humanoid or humanoid.Health <= 0 then return nil end
    return char, hrp, humanoid
end

-- Deteksi velocity mencurigakan: horizontal & ke atas dihitung,
-- jatuh ke bawah (gravitasi normal) TIDAK dihitung = tidak ada dodge palsu
local function isSuspiciousVelocity(vel, threshold)
    local horizontal = Vector3.new(vel.X, 0, vel.Z).Magnitude
    local upward = vel.Y
    return horizontal > threshold or upward > threshold
end

--// ================= NOCOLLIDE EVENT-BASED =================

local playerConnections = {}

local function disableCollision(part)
    if part:IsA("BasePart") and part.CanCollide then
        part.CanCollide = false
    end
end

-- Kembalikan collision pemain lain saat Anti Fling dimatikan
local function restoreCollision()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            for _, part in ipairs(plr.Character:GetDescendants()) do
                if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                    part.CanCollide = true
                end
            end
        end
    end
end

local function hookPlayerCharacter(plr, char)
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        disableCollision(part)
    end
    local conn = char.DescendantAdded:Connect(function(obj)
        if AntiFlingEnabled then
            task.defer(disableCollision, obj)
        end
    end)
    playerConnections[plr] = playerConnections[plr] or {}
    table.insert(playerConnections[plr], conn)
end

local function hookPlayer(plr)
    if plr == LocalPlayer then return end
    playerConnections[plr] = playerConnections[plr] or {}
    table.insert(playerConnections[plr], plr.CharacterAdded:Connect(function(char)
        if AntiFlingEnabled then
            task.wait(0.2)
            hookPlayerCharacter(plr, char)
        end
    end))
    if plr.Character then
        hookPlayerCharacter(plr, plr.Character)
    end
end

local function unhookPlayer(plr)
    if playerConnections[plr] then
        for _, conn in ipairs(playerConnections[plr]) do
            pcall(function() conn:Disconnect() end)
        end
        playerConnections[plr] = nil
    end
end

Players.PlayerAdded:Connect(function(plr)
    if AntiFlingEnabled then hookPlayer(plr) end
end)

Players.PlayerRemoving:Connect(unhookPlayer)

local function startNoCollideHooks()
    for _, plr in ipairs(Players:GetPlayers()) do
        if not playerConnections[plr] then
            hookPlayer(plr)
        end
    end
end

local function stopNoCollideHooks()
    for plr in pairs(playerConnections) do
        unhookPlayer(plr)
    end
    restoreCollision()
end

-- Sweep ringan (cadangan, jalan tiap 0.25 detik saja - hemat performa iPhone)
local function noCollideSweep()
    if tick() - lastNoCollideSweep < NOCOLLIDE_SWEEP then return end
    lastNoCollideSweep = tick()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            for _, part in ipairs(plr.Character:GetDescendants()) do
                disableCollision(part)
            end
        end
    end
end

--// ================= DETEKSI PENGGANGGU =================

local reportAttacker -- di-set setelah GUI dibuat

-- Deteksi pelaku pakai SKOR KECURIGAAN, bukan sekadar terdekat:
-- makin dekat + makin ngebut ke arahmu = makin curiga. Teman diam tidak salah tuduh.
local function findNearestAttacker()
    local _, hrp = getCharParts()
    if not hrp then return nil, nil end

    local best, bestScore, bestDist = nil, 0, nil
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local ohrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if ohrp then
                local d = (ohrp.Position - hrp.Position).Magnitude
                if d < ATTACKER_RADIUS and d > 0.1 then
                    local toMe = (hrp.Position - ohrp.Position).Unit
                    local approach = ohrp.AssemblyLinearVelocity:Dot(toMe)
                    local score = (ATTACKER_RADIUS - d) + math.max(approach, 0) * 0.5
                    if score > bestScore then
                        best, bestScore, bestDist = plr, score, d
                    end
                end
            end
        end
    end
    return best, bestDist
end

--// ================= VELOCITY LIMITER + ANTI ANCHOR =================

local function limitVelocity()
    local _, hrp = getCharParts()
    if not hrp then return end

    -- Anti Anchor Trap: kalau ada yang meng-anchor HRP-mu, lepaskan
    if hrp.Anchored then
        hrp.Anchored = false
    end

    local vel = hrp.AssemblyLinearVelocity
    if isSuspiciousVelocity(vel, MAX_VELOCITY) then
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        if DetectAttackerEnabled and reportAttacker then
            reportAttacker("Fling")
        end
    end

    local rotVel = hrp.AssemblyAngularVelocity
    if rotVel.Magnitude > 50 then
        hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end
end

--// ================= ANTI PUSH =================

local heavyProps = PhysicalProperties.new(100, 0.3, 0.5)
local defaultProps = {}

local function applyAntiPush(enable)
    local char = LocalPlayer.Character
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            if enable then
                if defaultProps[part] == nil then
                    defaultProps[part] = part.CustomPhysicalProperties
                end
                part.CustomPhysicalProperties = heavyProps
            else
                part.CustomPhysicalProperties = defaultProps[part]
                defaultProps[part] = nil
            end
        end
    end
end

--// ================= ANTI RAGDOLL =================

local function antiRagdoll()
    local _, _, humanoid = getCharParts()
    if not humanoid then return end

    if humanoid.PlatformStand then
        humanoid.PlatformStand = false
    end
    if humanoid.Sit then
        humanoid.Sit = false
        humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
    end
    pcall(function()
        humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
        humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
    end)
end

local function restoreRagdollStates()
    local _, _, humanoid = getCharParts()
    if not humanoid then return end
    pcall(function()
        humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
        humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
    end)
end

--// ================= ANTI CONSTRAINT =================

local BLOCKED_CLASSES = {
    "BodyVelocity", "BodyAngularVelocity", "BodyGyro", "BodyPosition",
    "BodyForce", "BodyThrust", "RocketPropulsion",
    "LinearVelocity", "AngularVelocity", "VectorForce", "Torque",
    "AlignPosition", "AlignOrientation", "RopeConstraint", "SpringConstraint"
}

local function isBlockedClass(obj)
    for _, className in ipairs(BLOCKED_CLASSES) do
        if obj:IsA(className) then
            return true
        end
    end
    return false
end

local function purgeConstraints()
    local char = LocalPlayer.Character
    if not char then return end
    for _, obj in ipairs(char:GetDescendants()) do
        if isBlockedClass(obj) then
            pcall(function() obj:Destroy() end)
        end
    end
end

--// ================= AUTO DODGE + ZIGZAG =================

local function findSafeSpot(hrp, minDist, maxDist)
    minDist = minDist or DODGE_MIN_DIST
    maxDist = maxDist or DODGE_MAX_DIST

    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = { LocalPlayer.Character }
    rayParams.FilterType = Enum.RaycastFilterType.Exclude

    for _ = 1, 10 do
        local angle = math.random() * math.pi * 2
        local dist = math.random(minDist, maxDist)
        local offset = Vector3.new(math.cos(angle) * dist, 0, math.sin(angle) * dist)
        local origin = hrp.Position + offset + Vector3.new(0, 25, 0)

        local result = Workspace:Raycast(origin, Vector3.new(0, -120, 0), rayParams)
        if result and result.Instance and result.Instance.CanCollide then
            return CFrame.new(result.Position + Vector3.new(0, 3.5, 0))
        end
    end
    return nil
end

local dodgeNotify -- di-set setelah GUI dibuat

-- Teleport dengan fallback PivotTo (jalan di karakter custom juga)
local function safeTeleport(char, hrp, targetCFrame)
    local ok = pcall(function()
        hrp.CFrame = targetCFrame
    end)
    if not ok then
        pcall(function()
            char:PivotTo(targetCFrame)
        end)
    end
end

local function teleportHop(minDist, maxDist)
    local char, hrp = getCharParts()
    if not char or not hrp then return false end

    local target = findSafeSpot(hrp, minDist, maxDist)
    if not target then return false end

    smokeEffect(hrp.Position)
    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    safeTeleport(char, hrp, target * CFrame.Angles(0, math.random() * math.pi * 2, 0))
    smokeEffect(hrp.Position)
    lastSafeCFrame = hrp.CFrame
    return true
end

local function doDodge(reason)
    if tick() - lastDodgeTime < DODGE_COOLDOWN then return end
    if tick() < spawnGraceUntil then return end -- jangan dodge saat baru spawn
    if zigzagRunning then return end            -- jangan tumpuk zigzag

    local char, hrp = getCharParts()
    if not char or not hrp then return end

    lastDodgeTime = tick()

    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

    -- Cari pelaku SEBELUM pindah (posisi masih dekat pelaku)
    if DetectAttackerEnabled and reportAttacker then
        reportAttacker(reason)
    end

    if ZigzagDodgeEnabled then
        zigzagRunning = true
        task.spawn(function()
            for i = 1, ZIGZAG_HOPS do
                pcall(teleportHop, 10, 20)
                if i < ZIGZAG_HOPS then
                    task.wait(ZIGZAG_DELAY)
                end
            end
            zigzagRunning = false
        end)
    else
        teleportHop(DODGE_MIN_DIST, DODGE_MAX_DIST)
    end

    if dodgeNotify then
        dodgeNotify(reason)
    end
end

--// ================= WATCHER CONSTRAINT =================

local constraintWatcher
local function watchConstraints()
    if constraintWatcher then constraintWatcher:Disconnect() end
    local char = LocalPlayer.Character
    if not char then return end
    constraintWatcher = char.DescendantAdded:Connect(function(obj)
        if isBlockedClass(obj) then
            if AntiConstraintEnabled then
                task.defer(function()
                    pcall(function() obj:Destroy() end)
                end)
            end
            if DetectAttackerEnabled and reportAttacker then
                task.defer(function()
                    pcall(reportAttacker, "Constraint jahat")
                end)
            end
            if AutoDodgeEnabled then
                task.defer(function()
                    pcall(doDodge, "Constraint jahat")
                end)
            end
        end
    end)
end

--// ================= SAFE POSITION =================

local function safePositionCheck()
    local char, hrp, humanoid = getCharParts()
    if not char or not hrp then return end

    local vel = hrp.AssemblyLinearVelocity

    if lastSafeCFrame then
        local dist = (hrp.Position - lastSafeCFrame.Position).Magnitude
        if dist > MAX_DISTANCE_JUMP and isSuspiciousVelocity(vel, MAX_VELOCITY) then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            safeTeleport(char, hrp, lastSafeCFrame)
            return
        end
    end

    local state = humanoid:GetState()
    if vel.Magnitude < 60
    and state ~= Enum.HumanoidStateType.Freefall
    and state ~= Enum.HumanoidStateType.Ragdoll
    and state ~= Enum.HumanoidStateType.FallingDown then
        lastSafeCFrame = hrp.CFrame
    end
end

--// ================= DETEKSI GANGGUAN (dodge trigger) =================

local function dodgeDetection()
    if tick() < spawnGraceUntil then return end -- grace period

    local char, hrp, humanoid = getCharParts()
    if not char or not hrp then return end

    -- Hanya velocity horizontal/ke atas yang memicu (jatuh normal = aman)
    if isSuspiciousVelocity(hrp.AssemblyLinearVelocity, DODGE_TRIGGER_VELOCITY)
    or hrp.AssemblyAngularVelocity.Magnitude > 40 then
        doDodge("Fling terdeteksi")
        return
    end

    local state = humanoid:GetState()
    if humanoid.PlatformStand
    or state == Enum.HumanoidStateType.Ragdoll
    or state == Enum.HumanoidStateType.FallingDown then
        doDodge("Ragdoll paksa")
        return
    end
end

local connection
local function startProtection()
    if connection then connection:Disconnect() end
    connection = RunService.Heartbeat:Connect(function()
        if AutoDodgeEnabled then
            pcall(dodgeDetection)
        end
        if AntiFlingEnabled then
            pcall(noCollideSweep)   -- sweep ringan tiap 0.25s (bukan tiap frame)
            pcall(limitVelocity)
        end
        if AntiRagdollEnabled then
            pcall(antiRagdoll)
        end
        if SafePositionEnabled then
            pcall(safePositionCheck)
        end
    end)
end

startProtection()
watchConstraints()

local notifyToken = 0
dodgeNotify = function(reason)
    notifyToken = notifyToken + 1
    local myToken = notifyToken
    NotifyLabel.Text = "Menghindar! [" .. reason .. "]"
    NotifyLabel.Visible = true
    task.delay(2.5, function()
        if notifyToken == myToken then
            NotifyLabel.Visible = false
        end
    end)
end

-- Fungsi lapor pelaku (dengan cooldown biar tidak spam)
reportAttacker = function(reason)
    if tick() - lastDetectTime < 1.5 then return end
    lastDetectTime = tick()

    local attacker, dist = findNearestAttacker()
    if attacker then
        AttackerLabel.Text = string.format("Pelaku: %s (%dm)", attacker.Name, math.floor(dist))
        AttackerLabel.TextColor3 = Color3.fromRGB(255, 140, 140)
        notifyToken = notifyToken + 1
        local myToken = notifyToken
        NotifyLabel.Text = string.format("[%s] Tersangka: %s (%d stud)", reason, attacker.Name, math.floor(dist))
        NotifyLabel.Visible = true
        task.delay(3, function()
            if notifyToken == myToken then
                NotifyLabel.Visible = false
            end
        end)
    else
        AttackerLabel.Text = "Pelaku: tidak terdeteksi"
        AttackerLabel.TextColor3 = Color3.fromRGB(200, 200, 210)
    end
end

--// ================= DRAG GUI (support layar sentuh) =================

local dragging = false
local dragInput, dragStart, startPos

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position

        local changedConn
        changedConn = input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                changedConn:Disconnect() -- putus koneksi, tidak bocor memori
            end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

--// ================= RESPAWN HANDLER =================

LocalPlayer.CharacterAdded:Connect(function(char)
    task.wait(1)
    lastSafeCFrame = nil
    lastDodgeTime = 0
    lastDetectTime = 0
    zigzagRunning = false
    spawnGraceUntil = tick() + SPAWN_GRACE -- grace period tiap respawn
    table.clear(defaultProps) -- buang referensi part karakter lama
    startProtection()
    watchConstraints()
    if AntiFlingEnabled then
        startNoCollideHooks()
    end
    if AntiPushEnabled then
        applyAntiPush(true)
    end
    if AntiConstraintEnabled then
        purgeConstraints()
    end
end)
